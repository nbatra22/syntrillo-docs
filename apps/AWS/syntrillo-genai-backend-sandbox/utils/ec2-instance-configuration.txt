# Install pip3
sudo apt update # <= important otherwise package retrieval error
sudo apt install python3-pip
sudo apt  install awscli
sudo apt install unzip

aws s3 cp s3://sandbox.syntrillo-clinic-backend.llm-server-packages/llm-service.zip .
unzip llm-service.zip
sudo pip install -r requirements.txt

sudo bash -c 'cat > /etc/systemd/system/flask_api.service << EOF
[Unit]
Description=Gunicorn instance to serve Flask API
After=network.target

[Service]
User=root
Group=root
WorkingDirectory=/home/ubuntu
# Environment="PATH=/home/ubuntu/testenv/bin"
#ExecStart=/home/ubuntu/testenv/bin/gunicorn --workers 2 --bind 0.0.0.0:5000 flask_api:app
#ExecStart=/home/ubuntu/.local/bin/gunicorn --workers 2 --bind 0.0.0.0:5000 flask_api:app
ExecStart=/usr/local/bin/gunicorn --workers 2 --bind 0.0.0.0:80 flask_api:app

[Install]
WantedBy=multi-user.target
EOF'

sudo systemctl daemon-reload
sudo systemctl start flask_api.service
sudo systemctl status flask_api.service
